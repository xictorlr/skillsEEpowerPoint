# Fase 3: Text — Plan de implementación

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Añadir la sección Text del panel: cambio de mayúsculas, tamaño de fuente ±, alineación de párrafo, alineación vertical, márgenes internos, autoajuste, ajuste de línea y limpieza de texto, sobre las formas con texto de la selección.

**Architecture:** Igual que las fases anteriores. `src/core/text.ts` es lógica pura sobre `TextShape`/`TextPatch`; `src/office/text.ts` carga y escribe `textFrame` (márgenes, autoajuste, wordWrap, alineación vertical), `textRange.text`, `textRange.font.size` y `textRange.paragraphFormat.horizontalAlignment` solo en formas con texto (misma detección por tipo y `containedType` de la Fase 2); nuevo tipo de comando `text`. Antes de añadir UI, `panel.ts` se parte en secciones (color, format, settings) con una interfaz `SectionContext`; la sección Text se añade como módulo propio.

**Tech Stack:** el existente. API: `TextFrame` y `TextRange` (PowerPointApi 1.4), `ParagraphFormat.horizontalAlignment` (1.4), `ShapeAutoSize`, `TextVerticalAlignment`, `ParagraphHorizontalAlignment`.

**Spec:** `docs/superpowers/specs/2026-10-14-slide-tools-design.md` § 5 Fase 3.

**Limitación de diseño (documentar en README):** Office.js no expone los "runs" de texto, así que cambiar mayúsculas y limpiar texto reescriben `textRange.text` completo; el texto conserva el formato del primer fragmento y pierde formatos mixtos dentro del cuadro. Cuando el tamaño de fuente es mixto (`font.size === null`), "Tamaño ±" omite esa forma. Las listas con viñetas conservan las viñetas al reescribir el texto, pero los saltos de párrafo deben preservarse: el texto llega con `\r`, `\n` o `\v` como separadores; las funciones de texto operan línea a línea y conservan el separador original.

```ts
/** Formato de texto de una forma con marco de texto. `null` = valor mixto. */
export interface Margins { left: number; right: number; top: number; bottom: number; }

export interface TextShape {
  id: string;
  text: string;
  fontSize: number | null;
  horizontalAlignment: string | null;   // PowerPoint.ParagraphHorizontalAlignment
  verticalAlignment: string;            // PowerPoint.TextVerticalAlignment
  autoSize: string;                     // PowerPoint.ShapeAutoSize
  wordWrap: boolean;
  margins: Margins;
}

export interface TextPatch {
  id: string;
  text?: string;
  fontSize?: number;
  horizontalAlignment?: string;
  verticalAlignment?: string;
  autoSize?: string;
  wordWrap?: boolean;
  margins?: Margins;
}

export type CaseMode = "upper" | "lower" | "sentence" | "title";
export type MarginPreset = "zero" | "standard";

export const ZERO_MARGINS: Margins = { left: 0, right: 0, top: 0, bottom: 0 };
/** Márgenes por defecto de PowerPoint: 0,1" lateral y 0,05" vertical. */
export const STANDARD_MARGINS: Margins = { left: 7.2, right: 7.2, top: 3.6, bottom: 3.6 };
export const MIN_FONT_SIZE = 1;
export const MAX_FONT_SIZE = 400;
const LOCALE = "es";

export interface Line { text: string; sep: string; }

/** Divide por separadores de párrafo/línea conservando cada separador. */
export function splitLines(text: string): Line[] {
  const lines: Line[] = [];
  let current = "";
  for (const ch of text) {
    if (ch === "\r" || ch === "\n" || ch === "\v") {
      lines.push({ text: current, sep: ch });
      current = "";
    } else current += ch;
  }
  lines.push({ text: current, sep: "" });
  return lines;
}

function joinLines(lines: Line[]): string {
  return lines.map((l) => l.text + l.sep).join("");
}

function sentenceCase(line: string): string {
  const lower = line.toLocaleLowerCase(LOCALE);
  let out = "";
  let capitalizeNext = true;
  for (const ch of lower) {
    if (capitalizeNext && /\p{L}/u.test(ch)) {
      out += ch.toLocaleUpperCase(LOCALE);
      capitalizeNext = false;
    } else {
      out += ch;
      if (ch === "." || ch === "!" || ch === "?") capitalizeNext = true;
    }
  }
  return out;
}

function titleCase(line: string): string {
  return line
    .toLocaleLowerCase(LOCALE)
    .replace(/(^|[\s(\["'¿¡-])(\p{L})/gu, (_m, pre: string, letter: string) => pre + letter.toLocaleUpperCase(LOCALE));
}

export function changeCase(text: string, mode: CaseMode): string {
  const lines = splitLines(text).map((l) => {
    if (mode === "upper") return { ...l, text: l.text.toLocaleUpperCase(LOCALE) };
    if (mode === "lower") return { ...l, text: l.text.toLocaleLowerCase(LOCALE) };
    if (mode === "sentence") return { ...l, text: sentenceCase(l.text) };
    return { ...l, text: titleCase(l.text) };
  });
  return joinLines(lines);
}

/** Colapsa espacios repetidos, recorta extremos de cada línea y elimina líneas vacías. */
export function cleanText(text: string): string {
  const lines = splitLines(text)
    .map((l) => ({ ...l, text: l.text.replace(/[ \t ]+/g, " ").trim() }))
    .filter((l) => l.text.length > 0);
  if (lines.length === 0) return "";
  const last = lines[lines.length - 1];
  if (last) last.sep = "";
  return joinLines(lines);
}

export function casePatches(shapes: TextShape[], mode: CaseMode): TextPatch[] {
  const out: TextPatch[] = [];
  for (const s of shapes) {
    const text = changeCase(s.text, mode);
    if (text !== s.text) out.push({ id: s.id, text });
  }
  return out;
}

export function cleanPatches(shapes: TextShape[]): TextPatch[] {
  const out: TextPatch[] = [];
  for (const s of shapes) {
    const text = cleanText(s.text);
    if (text !== s.text) out.push({ id: s.id, text });
  }
  return out;
}

/** Suma `delta` puntos al tamaño de fuente. Omite formas con tamaño mixto (null). */
export function fontSizePatches(shapes: TextShape[], delta: number): TextPatch[] {
  const out: TextPatch[] = [];
  for (const s of shapes) {
    if (s.fontSize === null) continue;
    const next = Math.min(MAX_FONT_SIZE, Math.max(MIN_FONT_SIZE, s.fontSize + delta));
    if (next !== s.fontSize) out.push({ id: s.id, fontSize: next });
  }
  return out;
}

export function paragraphAlignPatches(shapes: TextShape[], align: string): TextPatch[] {
  return shapes.filter((s) => s.horizontalAlignment !== align).map((s) => ({ id: s.id, horizontalAlignment: align }));
}

export function verticalAlignPatches(shapes: TextShape[], align: string): TextPatch[] {
  return shapes.filter((s) => s.verticalAlignment !== align).map((s) => ({ id: s.id, verticalAlignment: align }));
}

function sameMargins(a: Margins, b: Margins): boolean {
  return Math.abs(a.left - b.left) < 0.01 && Math.abs(a.right - b.right) < 0.01 && Math.abs(a.top - b.top) < 0.01 && Math.abs(a.bottom - b.bottom) < 0.01;
}

export function marginPatches(shapes: TextShape[], preset: MarginPreset): TextPatch[] {
  const target = preset === "zero" ? ZERO_MARGINS : STANDARD_MARGINS;
  return shapes.filter((s) => !sameMargins(s.margins, target)).map((s) => ({ id: s.id, margins: target }));
}

export function autoSizePatches(shapes: TextShape[], mode: string): TextPatch[] {
  return shapes.filter((s) => s.autoSize !== mode).map((s) => ({ id: s.id, autoSize: mode }));
}

export function wordWrapPatches(shapes: TextShape[], on: boolean): TextPatch[] {
  return shapes.filter((s) => s.wordWrap !== on).map((s) => ({ id: s.id, wordWrap: on }));
}
```

```ts
import type { TextPatch, TextShape } from "../core/text";
import type { FormatCaps } from "./format";

export const TEXT_FRAME_PROPS = "autoSizeSetting,wordWrap,verticalAlignment,leftMargin,rightMargin,topMargin,bottomMargin";
export const TEXT_RANGE_PROPS = "text";
export const PARAGRAPH_PROPS = "horizontalAlignment";

/** Encola la carga de texto en las formas con marco de texto (según caps.font). Requiere un sync posterior. */
export function loadTextProps(shapes: PowerPoint.Shape[], caps: Map<string, FormatCaps>): void {
  for (const s of shapes) {
    if (!caps.get(s.id)?.font) continue;
    s.textFrame.load(TEXT_FRAME_PROPS);
    s.textFrame.textRange.load(TEXT_RANGE_PROPS);
    s.textFrame.textRange.font.load("size");
    s.textFrame.textRange.paragraphFormat.load(PARAGRAPH_PROPS);
  }
}

export function toTextShape(shape: PowerPoint.Shape): TextShape {
  const tf = shape.textFrame;
  const tr = tf.textRange;
  return {
    id: shape.id,
    text: tr.text,
    fontSize: tr.font.size ?? null,
    horizontalAlignment: tr.paragraphFormat.horizontalAlignment == null ? null : String(tr.paragraphFormat.horizontalAlignment),
    verticalAlignment: String(tf.verticalAlignment),
    autoSize: String(tf.autoSizeSetting),
    wordWrap: tf.wordWrap,
    margins: { left: tf.leftMargin, right: tf.rightMargin, top: tf.topMargin, bottom: tf.bottomMargin },
  };
}

/** Escribe un parche de texto. No hace sync. */
export function applyTextPatch(shape: PowerPoint.Shape, p: TextPatch): void {
  const tf = shape.textFrame;
  if (p.text !== undefined) tf.textRange.text = p.text;
  if (p.fontSize !== undefined) tf.textRange.font.size = p.fontSize;
  if (p.horizontalAlignment !== undefined) tf.textRange.paragraphFormat.horizontalAlignment = p.horizontalAlignment as PowerPoint.ParagraphHorizontalAlignment;
  if (p.verticalAlignment !== undefined) tf.verticalAlignment = p.verticalAlignment as PowerPoint.TextVerticalAlignment;
  if (p.autoSize !== undefined) tf.autoSizeSetting = p.autoSize as PowerPoint.ShapeAutoSize;
  if (p.wordWrap !== undefined) tf.wordWrap = p.wordWrap;
  if (p.margins) {
    tf.leftMargin = p.margins.left;
    tf.rightMargin = p.margins.right;
    tf.topMargin = p.margins.top;
    tf.bottomMargin = p.margins.bottom;
  }
}
```
